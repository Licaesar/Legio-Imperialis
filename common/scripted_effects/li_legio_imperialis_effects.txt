# Amizades dos fundadores (chamado no historico de Kensoriel):
# cada Apostolo e amigo de Kensoriel, de Helaena e de todos os outros Apostolos.
li_founders_friendships_effect = {
	character:li_reid_astrea ?= { add_to_list = li_apostoli }
	character:li_ezio_auditore ?= { add_to_list = li_apostoli }
	character:li_emhyr_emreis ?= { add_to_list = li_apostoli }
	character:li_vistarios_granhieron ?= { add_to_list = li_apostoli }
	character:li_legarde_martyr ?= { add_to_list = li_apostoli }
	character:li_hrodric_gyliman ?= { add_to_list = li_apostoli }
	character:li_verso_lumere ?= { add_to_list = li_apostoli }
	character:li_altair_ramashiri ?= { add_to_list = li_apostoli }
	character:li_sigvardr_drakenfyr ?= { add_to_list = li_apostoli }
	character:li_miyamoto_musashi ?= { add_to_list = li_apostoli }
	character:li_aurelian_solerius ?= { add_to_list = li_apostoli }
	character:li_lakan_shaokan ?= { add_to_list = li_apostoli }
	character:li_astarinus_viriata ?= { add_to_list = li_apostoli }
	character:li_julius_euclius ?= { add_to_list = li_apostoli }
	character:li_radahn_varakh ?= { add_to_list = li_apostoli }
	character:li_alia_atreides ?= { add_to_list = li_apostoli }
	character:li_elessar_telcontar ?= { add_to_list = li_apostoli }

	every_in_list = {
		list = li_apostoli
		save_scope_as = li_apostle
		character:li_kensoriel_liphe ?= {
			if = {
				limit = { NOT = { has_relation_friend = scope:li_apostle } }
				set_relation_friend = { reason = friend_generic_history target = scope:li_apostle }
			}
		}
		character:li_helaena_undomiel ?= {
			if = {
				limit = { NOT = { has_relation_friend = scope:li_apostle } }
				set_relation_friend = { reason = friend_generic_history target = scope:li_apostle }
			}
		}
		every_in_list = {
			list = li_apostoli
			limit = {
				NOT = { this = scope:li_apostle }
				NOT = { has_relation_friend = scope:li_apostle }
			}
			set_relation_friend = { reason = friend_generic_history target = scope:li_apostle }
		}
	}
}
